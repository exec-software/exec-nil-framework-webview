find_package(Threads REQUIRED)
add_library(nilwebview_shared STATIC src/provider_shared.cpp)
target_compile_features(nilwebview_shared PUBLIC cxx_std_17)
target_include_directories(nilwebview_shared PUBLIC include src)
target_link_libraries(nilwebview_shared PUBLIC Threads::Threads)
set_target_properties(nilwebview_shared PROPERTIES
    CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)

    target_link_libraries(nilwebview_shared_tests PRIVATE nilwebview_shared)

    set_tests_properties(webview.script-transport PROPERTIES TIMEOUT 30)
endif()

if(NILWEBVIEW_BUILD_PROVIDER)
    if(WIN32)
        set(NILWEBVIEW_WEBVIEW2_SDK "" CACHE PATH "Verified extracted WebView2 SDK directory")
        if(NOT EXISTS "${NILWEBVIEW_WEBVIEW2_SDK}/build/native/include/WebView2.h")
            message(FATAL_ERROR "Run scripts/fetch-webview2.py and set NILWEBVIEW_WEBVIEW2_SDK to its output directory")
        endif()
        if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|arm64|aarch64)$")
            set(webview2_arch arm64)
        elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
            set(webview2_arch x64)
        else()
            message(FATAL_ERROR "This provider profile currently supports 64-bit Windows only")
        endif()
        set(webview2_loader "${NILWEBVIEW_WEBVIEW2_SDK}/build/native/${webview2_arch}/WebView2LoaderStatic.lib")
        if(NOT EXISTS "${webview2_loader}")
            message(FATAL_ERROR "Verified WebView2 SDK does not contain the selected architecture loader")
        endif()
        add_library(nilwebview SHARED src/provider_win32.cpp)
        target_include_directories(nilwebview PRIVATE "${NILWEBVIEW_WEBVIEW2_SDK}/build/native/include")
        target_link_libraries(nilwebview PRIVATE "${webview2_loader}" ole32 oleaut32 user32 shlwapi version advapi32 shell32 urlmon)
        target_compile_definitions(nilwebview PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    elseif(APPLE)
        enable_language(OBJCXX)
        add_library(nilwebview SHARED src/provider_macos.mm)
        set_source_files_properties(src/provider_macos.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
        set_target_properties(nilwebview PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
        target_link_libraries(nilwebview PRIVATE "-framework Cocoa" "-framework WebKit" "-framework CoreGraphics")
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.0)
        pkg_check_modules(WEBKIT6 REQUIRED IMPORTED_TARGET webkitgtk-6.0>=2.44)
        add_library(nilwebview SHARED src/provider_linux.cpp)
        target_link_libraries(nilwebview PRIVATE PkgConfig::GTK4 PkgConfig::WEBKIT6)
    else()
        message(FATAL_ERROR "No browser provider for this platform")
    endif()
    target_compile_definitions(nilwebview PRIVATE NILWEBVIEW_PROVIDER_BUILD)
    target_link_libraries(nilwebview PRIVATE nilwebview_shared)
    set_target_properties(nilwebview PROPERTIES
        CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
    install(TARGETS nilwebview LIBRARY DESTINATION providers RUNTIME DESTINATION providers)
endif()
